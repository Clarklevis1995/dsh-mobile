cmake_minimum_required(VERSION 3.5.0)
project(whaleness_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(KOTLIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated/${OHOS_ARCH}")
if(NOT EXISTS "${KOTLIN_DIR}/libwhaleness.so")
    message(FATAL_ERROR "Run ./scripts/build-harmony.sh native before building the Harmony app")
endif()
add_library(whaleness_kotlin SHARED IMPORTED)
set_target_properties(whaleness_kotlin PROPERTIES IMPORTED_LOCATION "${KOTLIN_DIR}/libwhaleness.so")
add_library(whaleness_bridge SHARED bridge.cpp)
target_include_directories(whaleness_bridge PRIVATE "${KOTLIN_DIR}")
target_link_libraries(whaleness_bridge PUBLIC whaleness_kotlin libace_napi.z.so)
